require('dotenv').config(); const TelegramBot = require('node-telegram-bot-api'); const User = require('./models/user'); const token = process.env.BOT_TOKEN; const bot = new TelegramBot(token, { polling: true }); const MINI_APP_URL = process.env.MINI_APP_URL || 'https://your-app-name.onrender.com'; const ADMIN_ID = process.env.ADMIN_ID || 'YOUR_TELEGRAM_ADMIN_ID'; const BANK_INFO = "0938331486 (Yohans)"; // ተጠቃሚው /start ሲሉ (ከሪፈራል እና ዌልካም ቦነስ ጋር) bot.onText(/\/start(?:\s+(.+))?/, async (msg, match) => { const chatId = msg.chat.id; const telegramId = msg.from.id.toString(); const firstName = msg.from.first_name || 'ተጫዋች'; const referrerId = match[1]; // የጋባዡ ID (ሪፈራል) let user = await User.findOne({ telegramId }); if (!user) { // አዲስ ተጠቃሚ ሲመዘገብ 10 ብር ዌልካም ቦነስ ይሰጠዋል user = await User.create({ telegramId, firstName, mainWallet: 10 }); // ሪፈራል ካለ ለጋባዡ 10 ብር ቦነስ ይጨመራል if (referrerId && referrerId !== telegramId) { await User.findOneAndUpdate( { telegramId: referrerId }, { $inc: { mainWallet: 10 } } ); } } const welcomeText = `👋 **Welcome to Yohans Bingo, ${firstName}!**\n\n🎁 አዲስ በመመዝገብዎ **10 ብር ዌልካም ቦነስ** አግኝተዋል!\n\nእውነተኛውን የቢንጎ ጨዋታ ከታች ባሉት አማራጮች ይጀምሩ።`; bot.sendMessage(chatId, welcomeText, { parse_mode: 'Markdown', reply_markup: { inline_keyboard: [ [{ text: '🎮 Play Yohans Bingo', web_app: { url: MINI_APP_URL } }, { text: '💰 Check Balance', callback_data: 'balance' }], [{ text: '💳 Deposit', callback_data: 'deposit' }, { text: '🔗 Invite', callback_data: 'invite' }] ] } }); }); // የቁልፍ ምላሾች (Callback Queries) bot.on('callback_query', async (query) => { const chatId = query.message.chat.id; const telegramId = query.from.id.toString(); const data = query.data; let user = await User.findOne({ telegramId }); if (data === 'balance') { const balance = user ? user.mainWallet : 0; bot.sendMessage(chatId, `💰 የእርስዎ ቀሪ ሂሳብ: ${balance} ብር ነው`); } else if (data === 'deposit') { bot.sendMessage(chatId, `💳 **Yohans Bingo Deposit**\n\nእባክዎ ብሩን ወደዚህ የቴሌብር ቁጥር ይላኩ:\n👉 **${BANK_INFO}**\n\nከላኩ በኋላ የግብይት ማረጋገጫውን ለአድሚን ያጋሩ።`); } else if (data === 'invite') { bot.sendMessage(chatId, `🔗 **Invite Friends & Earn**\n\nጓደኛዎችዎን በመጋበዝ 10 ብር ቦነስ ያግኙ!\nየእርስዎ መጋበዣ ሊንክ:\nt.me/your_bot_name?start=${telegramId}`); } bot.answerCallbackQuery(query.id); }); // አድሚን ብር ለመጨመር: /add [telegramId] [amount] bot.onText(/\/add (\d+) (\d+)/, async (msg, match) => { const chatId = msg.chat.id; const senderId = msg.from.id.toString(); if (senderId !== ADMIN_ID && ADMIN_ID !== 'YOUR_TELEGRAM_ADMIN_ID') { return bot.sendMessage(chatId, "❌ ይህንን ትዕዛዝ መጠቀም የሚችሉት አድሚኖች ብቻ ናቸው!"); } const targetTelegramId = match[1]; const amount = Number(match[2]); let user = await User.findOne({ telegramId: targetTelegramId }); if (user) { user.mainWallet += amount; await user.save(); bot.sendMessage(chatId, `✅ ተሳክቷል! ለተጠቃሚ ${targetTelegramId} መጠን ${amount} ብር ተጭኗል። አጠቃላይ ሂሳብ: ${user.mainWallet} ብር`); } else { bot.sendMessage(chatId, "❌ ተጠቃሚው በዳታቤዝ አልተገኘም!"); } }); // አድሚን ብር ለመቀነስ/ለማውጣት: /deduct [telegramId] [amount] bot.onText(/\/deduct (\d+) (\d+)/, async (msg, match) => { const chatId = msg.chat.id; const senderId = msg.from.id.toString(); if (senderId !== ADMIN_ID && ADMIN_ID !== 'YOUR_TELEGRAM_ADMIN_ID') { return bot.sendMessage(chatId, "❌ ይህንን ትዕዛዝ መጠቀም የሚችሉት አድሚኖች ብቻ ናቸው!"); } const targetTelegramId = match[1]; const amount = Number(match[2]); let user = await User.findOne({ telegramId: targetTelegramId }); if (user) { user.mainWallet = Math.max(0, user.mainWallet - amount); await user.save(); bot.sendMessage(chatId, `✅ ተሳክቷል! ከተጠቃሚ ${targetTelegramId} ላይ ${amount} ብር ቀንሷል። አጠቃላይ ቀሪ ሂሳብ: ${user.mainWallet} ብር`); } else { bot.sendMessage(chatId, "❌ ተጠቃሚው በዳታቤዝ አልተገኘም!"); } }); console.log('Yohans Bingo Bot is fully active with all features...');
